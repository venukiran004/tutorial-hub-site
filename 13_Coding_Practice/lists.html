def merge_sorted(a, b):
    result = []
    i = j = 0
    while i < len(a) and j < len(b):
        if a[i] <= b[j]:
            result.append(a[i]); i += 1
        else:
            result.append(b[j]); j += 1
    result.extend(a[i:])
    result.extend(b[j:])
    return result

def second_largest(lst):
    if len(lst) < 2:
        return None
    first = second = float('-inf')
    for num in lst:
        if num > first:
            second = first
            first = num
        elif num > second and num != first:
            second = num
    return second if second != float('-inf') else None

Program 41: Find Consecutive Sequences

def consecutive_sequences(lst):
    if not lst:
        return []
    lst = sorted(set(lst))
    sequences = []
    current = [lst[0]]
    for i in range(1, len(lst)):
        if lst[i] == lst[i-1] + 1:
            current.append(lst[i])
        else:
            sequences.append(current)
            current = [lst[i]]
    sequences.append(current)
    return sequences

Program 49: Sliding Window Maximum

from collections import deque

def sliding_window_max(nums, k):
    dq = deque()  # stores indices
    result = []
    for i in range(len(nums)):
        while dq and dq[0] < i - k + 1:
            dq.popleft()
        while dq and nums[dq[-1]] < nums[i]:
            dq.pop()
        dq.append(i)
        if i >= k - 1:
            result.append(nums[dq[0]])
    return result

print(sliding_window_max([1, 3, -1, -3, 5, 3, 6, 7], 3))

def sliding_avg(nums, k):
    if len(nums) < k:
        return []
    window_sum = sum(nums[:k])

Program 52: Kadane's Algorithm (Max Subarray Sum)

def max_subarray_sum(nums):
    max_sum = current = nums[0]
    start = end = temp_start = 0
    for i in range(1, len(nums)):
        if nums[i] > current + nums[i]:
            current = nums[i]
            temp_start = i
        else:
            current += nums[i]
        if current > max_sum:
            max_sum = current
            start = temp_start
            end = i
    return max_sum, nums[start:end+1]

Program 53: Find Equilibrium Index

def equilibrium_index(lst):
    """Index where sum of left == sum of right."""
    total = sum(lst)
    left_sum = 0
    for i in range(len(lst)):
        right_sum = total - left_sum - lst[i]
        if left_sum == right_sum:
            return i
        left_sum += lst[i]
    return -1

Program 55: Trapping Rain Water

def trap_water(heights):
    left, right = 0, len(heights) - 1
    left_max = right_max = water = 0
    while left < right:
        if heights[left] < heights[right]:
            left_max = max(left_max, heights[left])
            water += left_max - heights[left]
            left += 1
        else:
            right_max = max(right_max, heights[right])
            water += right_max - heights[right]
            right -= 1
    return water

def max_water(heights):
    left, right = 0, len(heights) - 1
    max_area = 0

def two_sum(nums, target):

Program 172: Implement LFU Cache Using Lists

from collections import defaultdict, OrderedDict

class LFUCache:
    def __init__(self, capacity):
        self.cap = capacity
        self.min_freq = 0
        self.key_val = {}
        self.key_freq = {}
        self.freq_keys = defaultdict(OrderedDict)
    
    def get(self, key):
        if key not in self.key_val: return -1
        self._update_freq(key)
        return self.key_val[key]
    
    def put(self, key, val):
        if self.cap <= 0: return
        if key in self.key_val:
            self.key_val[key] = val
            self._update_freq(key)
            return
        if len(self.key_val) >= self.cap:
            evict_key, _ = self.freq_keys[self.min_freq].popitem(last=False)
            del self.key_val[evict_key]
            del self.key_freq[evict_key]
        self.key_val[key] = val
        self.key_freq[key] = 1
        self.freq_keys[1][key] = None
        self.min_freq = 1
    
    def _update_freq(self, key):
        freq = self.key_freq[key]
        del self.freq_keys[freq][key]
        if not self.freq_keys[freq] and freq == self.min_freq:
            self.min_freq += 1
        self.key_freq[key] = freq + 1
        self.freq_keys[freq + 1][key] = None

class CircularBuffer:
    def __init__(self, capacity):
        self.buf = [None] * capacity
        self.cap = capacity
        self.head = self.size = 0
    
    def push(self, val):
        idx = (self.head + self.size) % self.cap
        if self.size == self.cap:
            self.head = (self.head + 1) % self.cap
        else:
            self.size += 1
        self.buf[idx] = val
    
    def pop(self):
        if self.size == 0: return None
        val = self.buf[self.head]
        self.head = (self.head + 1) % self.cap
        self.size -= 1
        return val
    
    def to_list(self):
        return [self.buf[(self.head + i) % self.cap] for i in range(self.size)]

Program 210: Merge two sorted lists

"""
Merge two already sorted lists into one sorted list.
Two-pointer technique.
Time Complexity: O(n + m)
Space Complexity: O(n + m)
"""


def merge_sorted_lists(lst1, lst2):
    """Merge two sorted lists using two pointers."""
    result = []
    i, j = 0, 0
    while i < len(lst1) and j < len(lst2):
        if lst1[i] <= lst2[j]:
            result.append(lst1[i])
            i += 1
        else:
            result.append(lst2[j])
            j += 1
    # Append remaining elements
    result.extend(lst1[i:])
    result.extend(lst2[j:])
    return result

Program 213: Flatten a nested list

Convert a nested list (any depth) into a single flat list.
Approach: Recursion to handle arbitrary nesting.
Time Complexity: O(n) where n is total number of elements
"""


def flatten(lst):
    """Recursively flatten a nested list."""
    result = []
    for item in lst:
        if isinstance(item, list):
            result.extend(flatten(item))
        else:
            result.append(item)
    return result


def flatten_iterative(lst):
    """Iterative flattening using a stack."""
    stack = list(lst)
    result = []
    while stack:
        item = stack.pop(0)
        if isinstance(item, list):
            stack = item + stack
        else:
            result.append(item)
    return result
