class BSTNode:
    def __init__(self, val):

        self.left = None
        self.right = None

class BST:
    def __init__(self):
        self.root = None
    
    def insert(self, val):
        self.root = self._insert(self.root, val)
    
    def _insert(self, node, val):
        if not node:
            return BSTNode(val)
        if val < node.val:
            node.left = self._insert(node.left, val)
        elif val > node.val:
            node.right = self._insert(node.right, val)
        return node
    
    def search(self, val):
        return self._search(self.root, val)
    
    def _search(self, node, val):
        if not node or node.val == val:
            return node
        if val < node.val:
            return self._search(node.left, val)
        return self._search(node.right, val)
    
    def delete(self, val):
        self.root = self._delete(self.root, val)
    
    def _delete(self, node, val):
        if not node:
            return None
        if val < node.val:
            node.left = self._delete(node.left, val)
        elif val > node.val:
            node.right = self._delete(node.right, val)
        else:
            if not node.left:
                return node.right
            if not node.right:
                return node.left
            successor = self._min_node(node.right)
            node.val = successor.val
            node.right = self._delete(node.right, successor.val)
        return node
    
    def _min_node(self, node):
        while node.left:
            node = node.left
        return node
    
    def inorder(self):
        result = []
        self._inorder(self.root, result)
        return result
    
    def _inorder(self, node, result):
        if node:
            self._inorder(node.left, result)
            result.append(node.val)
            self._inorder(node.right, result)

Program 2: AVL Tree (Self-Balancing BST)

class AVLNode:
    def __init__(self, val):
        self.val = val
        self.left = None
        self.right = None
        self.height = 1

class AVLTree:
    def _height(self, node):
        return node.height if node else 0
    
    def _balance(self, node):
        return self._height(node.left) - self._height(node.right) if node else 0
    
    def _rotate_right(self, y):
        x = y.left
        t = x.right
        x.right = y
        y.left = t
        y.height = max(self._height(y.left), self._height(y.right)) + 1
        x.height = max(self._height(x.left), self._height(x.right)) + 1
        return x
    
    def _rotate_left(self, x):
        y = x.right
        t = y.left
        y.left = x
        x.right = t
        x.height = max(self._height(x.left), self._height(x.right)) + 1
        y.height = max(self._height(y.left), self._height(y.right)) + 1
        return y
    
    def insert(self, root, val):
        if not root:
            return AVLNode(val)
        if val < root.val:
            root.left = self.insert(root.left, val)
        else:
            root.right = self.insert(root.right, val)
        
        root.height = max(self._height(root.left), self._height(root.right)) + 1
        balance = self._balance(root)
        
        if balance > 1 and val < root.left.val:
            return self._rotate_right(root)
        if balance < -1 and val > root.right.val:
            return self._rotate_left(root)
        if balance > 1 and val > root.left.val:
            root.left = self._rotate_left(root.left)
            return self._rotate_right(root)
        if balance < -1 and val < root.right.val:
            root.right = self._rotate_right(root.right)
            return self._rotate_left(root)
        
        return root

Program 3: Segment Tree (Range Sum Query)

class SegmentTree:
    def __init__(self, arr):
        self.n = len(arr)
        self.tree = [0] * (4 * self.n)
        self._build(arr, 1, 0, self.n - 1)
    
    def _build(self, arr, node, start, end):
        if start == end:
            self.tree[node] = arr[start]
            return
        mid = (start + end) // 2
        self._build(arr, 2*node, start, mid)
        self._build(arr, 2*node+1, mid+1, end)
        self.tree[node] = self.tree[2*node] + self.tree[2*node+1]
    
    def query(self, l, r):
        return self._query(1, 0, self.n-1, l, r)
    
    def _query(self, node, start, end, l, r):
        if r < start or end < l:
            return 0
        if l <= start and end <= r:
            return self.tree[node]
        mid = (start + end) // 2
        return (self._query(2*node, start, mid, l, r) + 
                self._query(2*node+1, mid+1, end, l, r))
    
    def update(self, idx, val):
        self._update(1, 0, self.n-1, idx, val)
    
    def _update(self, node, start, end, idx, val):
        if start == end:
            self.tree[node] = val
            return
        mid = (start + end) // 2
        if idx <= mid:
            self._update(2*node, start, mid, idx, val)
        else:
            self._update(2*node+1, mid+1, end, idx, val)
        self.tree[node] = self.tree[2*node] + self.tree[2*node+1]

st.update(2, 10)  # Change 5 to 10
print(f"Sum [1,3] after update: {st.query(1, 3)}")  # 3+10+7 = 20

Program 4: Fenwick Tree (Binary Indexed Tree)

class FenwickTree:
    def __init__(self, n):
        self.n = n
        self.tree = [0] * (n + 1)
    
    def update(self, i, delta):
        i += 1  # 1-indexed
        while i <= self.n:
            self.tree[i] += delta
            i += i & (-i)
    
    def prefix_sum(self, i):
        i += 1
        s = 0
        while i > 0:
            s += self.tree[i]
            i -= i & (-i)
        return s
    
    def range_sum(self, l, r):
        return self.prefix_sum(r) - (self.prefix_sum(l-1) if l > 0 else 0)

Program 5: Graph — Kruskal's MST

class KruskalMST:
    def __init__(self, vertices):
        self.vertices = vertices
        self.parent = {v: v for v in vertices}
        self.rank = {v: 0 for v in vertices}
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        return True
    
    def mst(self, edges):
        edges.sort(key=lambda x: x[2])
        result = []
        total_weight = 0
        for u, v, w in edges:
            if self.union(u, v):
                result.append((u, v, w))
                total_weight += w
        return result, total_weight

Program 6: Prim's MST

import heapq

def prim_mst(graph, start):
    visited = set()
    mst = []
    edges = [(0, start, None)]  # (weight, node, from)
    total = 0
    
    while edges and len(visited) < len(graph):
        weight, node, from_node = heapq.heappop(edges)
        if node in visited:
            continue
        visited.add(node)
        if from_node is not None:
            mst.append((from_node, node, weight))
            total += weight
        for neighbor, w in graph[node]:
            if neighbor not in visited:
                heapq.heappush(edges, (w, neighbor, node))
    
    return mst, total

    'A': [('B', 4), ('C', 2)],
    'B': [('A', 4), ('C', 1), ('D', 5)],
    'C': [('A', 2), ('B', 1), ('D', 8), ('E', 10)],
    'D': [('B', 5), ('C', 8), ('E', 2)],
    'E': [('C', 10), ('D', 2)]
}

mst, total = prim_mst(graph, 'A')
print(f"MST: {mst}")
print(f"Total: {total}")  # 9

Program 7: Floyd-Warshall (All Pairs Shortest Path)

def floyd_warshall(graph, n):
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    
    for i in range(n):
        dist[i][i] = 0
    
    for u, v, w in graph:
        dist[u][v] = w
    
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    return dist

Program 8: Counting Sort

def counting_sort(arr):
    if not arr:
        return arr
    max_val = max(arr)
    min_val = min(arr)
    range_size = max_val - min_val + 1
    
    count = [0] * range_size
    output = [0] * len(arr)
    
    for num in arr:
        count[num - min_val] += 1
    
    for i in range(1, len(count)):
        count[i] += count[i - 1]
    
    for num in reversed(arr):
        output[count[num - min_val] - 1] = num
        count[num - min_val] -= 1
    
    return output

print(counting_sort([4, 2, 2, 8, 3, 3, 1]))
# Output: [1, 2, 2, 3, 3, 4, 8]

Program 9: Radix Sort

def radix_sort(arr):
    if not arr:
        return arr
    max_digits = len(str(max(arr)))
    
    for digit in range(max_digits):
        buckets = [[] for _ in range(10)]
        for num in arr:
            d = (num // (10 ** digit)) % 10
            buckets[d].append(num)
        arr = [num for bucket in buckets for num in bucket]
    
    return arr

Program 10: Shell Sort

def shell_sort(arr):
    n = len(arr)
    gap = n // 2
    while gap > 0:
        for i in range(gap, n):
            temp = arr[i]
            j = i
            while j >= gap and arr[j - gap] > temp:
                arr[j] = arr[j - gap]
                j -= gap
            arr[j] = temp
        gap //= 2
    return arr

print(shell_sort([12, 34, 54, 2, 3]))
# Output: [2, 3, 12, 34, 54]

Program 11: Tim Sort (Simplified)

def insertion_sort_range(arr, left, right):
    for i in range(left + 1, right + 1):
        key = arr[i]
        j = i - 1
        while j >= left and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key

def merge_sorted(arr, l, m, r):
    left = arr[l:m+1]
    right = arr[m+1:r+1]
    i = j = 0
    k = l
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            arr[k] = left[i]
            i += 1
        else:
            arr[k] = right[j]
            j += 1
        k += 1
    while i < len(left):
        arr[k] = left[i]
        i += 1; k += 1
    while j < len(right):
        arr[k] = right[j]
        j += 1; k += 1

def tim_sort(arr):
    RUN = 32
    n = len(arr)
    for start in range(0, n, RUN):
        end = min(start + RUN - 1, n - 1)
        insertion_sort_range(arr, start, end)
    
    size = RUN
    while size < n:
        for left in range(0, n, 2 * size):
            mid = min(left + size - 1, n - 1)
            right = min(left + 2 * size - 1, n - 1)
            if mid < right:
                merge_sorted(arr, left, mid, right)
        size *= 2
    return arr

Program 12: Longest Increasing Subsequence

from bisect import bisect_left

def lis_dp(nums):
    """O(n²) DP approach"""
    if not nums:
        return 0
    dp = [1] * len(nums)
    for i in range(1, len(nums)):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)

def lis_binary(nums):
    """O(n log n) approach with binary search"""
    tails = []
    for num in nums:
        idx = bisect_left(tails, num)
        if idx == len(tails):
            tails.append(num)
        else:
            tails[idx] = num
    return len(tails)

nums = [10, 9, 2, 5, 3, 7, 101, 18]
print(f"LIS length (DP): {lis_dp(nums)}")        # 4
print(f"LIS length (Binary): {lis_binary(nums)}")  # 4

Program 13: Longest Palindromic Substring

def longest_palindrome(s):
    def expand(left, right):
        while left >= 0 and right < len(s) and s[left] == s[right]:
            left -= 1
            right += 1
        return s[left+1:right]
    
    result = ""
    for i in range(len(s)):
        # Odd length
        p1 = expand(i, i)
        # Even length
        p2 = expand(i, i + 1)
        
        if len(p1) > len(result):
            result = p1
        if len(p2) > len(result):
            result = p2
    
    return result

print(longest_palindrome("cbbd"))     # "bb"

Program 14: Word Break Problem

def word_break(s, word_dict):
    n = len(s)
    dp = [False] * (n + 1)
    dp[0] = True
    
    for i in range(1, n + 1):
        for j in range(i):
            if dp[j] and s[j:i] in word_dict:
                dp[i] = True
                break
    
    return dp[n]

def word_break_all(s, word_dict):
    """Return all possible segmentations"""
    memo = {}
    def backtrack(start):
        if start in memo:
            return memo[start]
        if start == len(s):
            return [[]]
        results = []
        for end in range(start + 1, len(s) + 1):
            word = s[start:end]
            if word in word_dict:
                for rest in backtrack(end):
                    results.append([word] + rest)
        memo[start] = results
        return results
    
    return [" ".join(words) for words in backtrack(0)]

word_dict = {"leet", "code", "leetcode"}
print(word_break("leetcode", word_dict))  # True

print(word_break_all("catsanddog", word_dict2))
# ['cats and dog', 'cat sand dog']

Program 15: Matrix Chain Multiplication

def matrix_chain_order(dims):
    n = len(dims) - 1
    dp = [[0] * n for _ in range(n)]
    
    for chain_len in range(2, n + 1):
        for i in range(n - chain_len + 1):
            j = i + chain_len - 1
            dp[i][j] = float('inf')
            for k in range(i, j):
                cost = dp[i][k] + dp[k+1][j] + dims[i] * dims[k+1] * dims[j+1]
                dp[i][j] = min(dp[i][j], cost)
    
    return dp[0][n-1]

Program 16: Rod Cutting Problem

def rod_cutting(prices, n):
    dp = [0] * (n + 1)
    cuts = [0] * (n + 1)
    
    for i in range(1, n + 1):
        for j in range(1, i + 1):
            if j <= len(prices) and prices[j-1] + dp[i-j] > dp[i]:
                dp[i] = prices[j-1] + dp[i-j]
                cuts[i] = j
    
    # Reconstruct solution
    pieces = []
    remaining = n
    while remaining > 0:
        pieces.append(cuts[remaining])
        remaining -= cuts[remaining]
    
    return dp[n], pieces

max_revenue, pieces = rod_cutting(prices, 8)

Program 17: Egg Drop Problem

def egg_drop(eggs, floors):
    dp = [[0] * (floors + 1) for _ in range(eggs + 1)]
    
    for i in range(1, eggs + 1):
        dp[i][0] = 0
        dp[i][1] = 1
    
    for j in range(1, floors + 1):
        dp[1][j] = j
    
    for i in range(2, eggs + 1):
        for j in range(2, floors + 1):
            dp[i][j] = float('inf')
            for k in range(1, j + 1):
                cost = 1 + max(dp[i-1][k-1], dp[i][j-k])
                dp[i][j] = min(dp[i][j], cost)
    
    return dp[eggs][floors]

Program 18: Trie — Word Search

class TrieNode:
    def __init__(self):
        self.children = {}
        self.word = None

def word_search_ii(board, words):
    root = TrieNode()
    for word in words:
        node = root
        for c in word:
            node = node.children.setdefault(c, TrieNode())
        node.word = word
    
    rows, cols = len(board), len(board[0])
    found = []
    
    def dfs(r, c, node):
        if r < 0 or r >= rows or c < 0 or c >= cols:
            return
        char = board[r][c]
        if char not in node.children:
            return
        
        next_node = node.children[char]
        if next_node.word:
            found.append(next_node.word)
            next_node.word = None  # Avoid duplicates
        
        board[r][c] = '#'
        for dr, dc in [(0,1),(0,-1),(1,0),(-1,0)]:
            dfs(r+dr, c+dc, next_node)
        board[r][c] = char
    
    for r in range(rows):
        for c in range(cols):
            dfs(r, c, root)
    
    return found

Program 19: Graph Coloring

def graph_coloring(graph, num_colors):
    n = len(graph)
    colors = [-1] * n
    
    def is_safe(node, color):
        for neighbor in graph[node]:
            if colors[neighbor] == color:
                return False
        return True
    
    def solve(node):
        if node == n:
            return True
        for color in range(num_colors):
            if is_safe(node, color):
                colors[node] = color
                if solve(node + 1):
                    return True
                colors[node] = -1
        return False
    
    if solve(0):
        return colors
    return None

Program 20: Detect Cycle in Directed Graph

def has_cycle_directed(graph, n):
    WHITE, GRAY, BLACK = 0, 1, 2
    color = [WHITE] * n
    
    def dfs(node):
        color[node] = GRAY
        for neighbor in graph.get(node, []):
            if color[neighbor] == GRAY:
                return True
            if color[neighbor] == WHITE and dfs(neighbor):
                return True
        color[node] = BLACK
        return False
    
    return any(color[i] == WHITE and dfs(i) for i in range(n))

# Graph with cycle: 0→1→2→0

Program 21: Strongly Connected Components (Tarjan's)

def tarjan_scc(graph, n):
    index_counter = [0]
    stack = []
    lowlink = [0] * n
    index = [0] * n
    on_stack = [False] * n
    visited = [False] * n
    sccs = []
    
    def strongconnect(v):
        index[v] = index_counter[0]
        lowlink[v] = index_counter[0]
        index_counter[0] += 1
        visited[v] = True
        stack.append(v)
        on_stack[v] = True
        
        for w in graph.get(v, []):
            if not visited[w]:
                strongconnect(w)
                lowlink[v] = min(lowlink[v], lowlink[w])
            elif on_stack[w]:
                lowlink[v] = min(lowlink[v], index[w])
        
        if lowlink[v] == index[v]:
            scc = []
            while True:
                w = stack.pop()
                on_stack[w] = False
                scc.append(w)
                if w == v:
                    break
            sccs.append(scc)
    
    for v in range(n):
        if not visited[v]:
            strongconnect(v)
    
    return sccs

graph = {0: [1], 1: [2], 2: [0, 3], 3: [4], 4: [5], 5: [3]}
print(tarjan_scc(graph, 6))
# Output: [[4, 5, 3], [0, 2, 1]] — two SCCs

def rabin_karp(text, pattern, prime=101):
    n, m = len(text), len(pattern)
    d = 256  # Number of characters
    
    pattern_hash = 0
    text_hash = 0
    h = pow(d, m - 1, prime)
    matches = []
    
    # Calculate hash of pattern and first window
    for i in range(m):
        pattern_hash = (d * pattern_hash + ord(pattern[i])) % prime
        text_hash = (d * text_hash + ord(text[i])) % prime
    
    for i in range(n - m + 1):
        if pattern_hash == text_hash:
            if text[i:i+m] == pattern:  # Verify
                matches.append(i)
        
        if i < n - m:
            text_hash = (d * (text_hash - ord(text[i]) * h) + ord(text[i + m])) % prime
            if text_hash < 0:
                text_hash += prime
    
    return matches

def tsp(dist):
    n = len(dist)
    INF = float('inf')
    dp = [[INF] * n for _ in range(1 << n)]
    dp[1][0] = 0
    
    for mask in range(1 << n):
        for u in range(n):
            if dp[mask][u] == INF:
                continue
            for v in range(n):
                if mask & (1 << v):
                    continue
                new_mask = mask | (1 << v)
                dp[new_mask][v] = min(dp[new_mask][v], dp[mask][u] + dist[u][v])
    
    full_mask = (1 << n) - 1
    return min(dp[full_mask][i] + dist[i][0] for i in range(1, n))

def convex_hull(points):
    def cross(o, a, b):
        return (a[0]-o[0])*(b[1]-o[1]) - (a[1]-o[1])*(b[0]-o[0])
    
    points = sorted(set(points))
    if len(points) <= 1:
        return points
    
    # Lower hull
    lower = []
    for p in points:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    
    # Upper hull
    upper = []
    for p in reversed(points):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    
    return lower[:-1] + upper[:-1]

import math

class LCATree:
    def __init__(self, n):
        self.n = n
        self.LOG = int(math.log2(n)) + 1
        self.up = [[0] * n for _ in range(self.LOG)]
        self.depth = [0] * n
        self.adj = [[] for _ in range(n)]
    
    def add_edge(self, u, v):
        self.adj[u].append(v)
        self.adj[v].append(u)
    
    def build(self, root=0):
        from collections import deque
        visited = [False] * self.n
        queue = deque([root])
        visited[root] = True
        
        while queue:
            u = queue.popleft()
            for v in self.adj[u]:
                if not visited[v]:
                    visited[v] = True
                    self.depth[v] = self.depth[u] + 1
                    self.up[0][v] = u
                    queue.append(v)
        
        for k in range(1, self.LOG):
            for v in range(self.n):
                self.up[k][v] = self.up[k-1][self.up[k-1][v]]
    
    def lca(self, u, v):
        if self.depth[u] < self.depth[v]:
            u, v = v, u
        diff = self.depth[u] - self.depth[v]
        for k in range(self.LOG):
            if (diff >> k) & 1:
                u = self.up[k][u]
        if u == v:
            return u
        for k in range(self.LOG - 1, -1, -1):
            if self.up[k][u] != self.up[k][v]:
                u = self.up[k][u]
                v = self.up[k][v]
        return self.up[0][u]

        return best, best_dist

Program 49: Priority Queue with Decrease Key

import heapq

class IndexedPQ:
    def __init__(self):
        self.heap = []
        self.entry_finder = {}
        self.counter = 0
        self.REMOVED = '<removed>'
    
    def push(self, key, priority):
        if key in self.entry_finder:
            self.remove(key)
        entry = [priority, self.counter, key]
        self.counter += 1
        self.entry_finder[key] = entry
        heapq.heappush(self.heap, entry)
    
    def remove(self, key):
        entry = self.entry_finder.pop(key)
        entry[-1] = self.REMOVED
    
    def pop(self):
        while self.heap:
            priority, _, key = heapq.heappop(self.heap)
            if key is not self.REMOVED:
                del self.entry_finder[key]
                return key, priority
        raise KeyError("Empty priority queue")
    
    def decrease_key(self, key, new_priority):
        self.push(key, new_priority)
    
    def __len__(self):
        return len(self.entry_finder)

Program 81: LFU Cache

from collections import defaultdict, OrderedDict

class LFUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}          # key -> (value, freq)
        self.freq_map = defaultdict(OrderedDict)  # freq -> OrderedDict of keys
        self.min_freq = 0
    
    def get(self, key):
        if key not in self.cache:
            return -1
        value, freq = self.cache[key]
        del self.freq_map[freq][key]
        if not self.freq_map[freq]:
            del self.freq_map[freq]
            if self.min_freq == freq:
                self.min_freq += 1
        self.cache[key] = (value, freq + 1)
        self.freq_map[freq + 1][key] = None
        return value
    
    def put(self, key, value):
        if self.capacity <= 0:
            return
        if key in self.cache:
            self.cache[key] = (value, self.cache[key][1])
            self.get(key)
            return
        if len(self.cache) >= self.capacity:
            evict_key, _ = self.freq_map[self.min_freq].popitem(last=False)
            if not self.freq_map[self.min_freq]:
                del self.freq_map[self.min_freq]
            del self.cache[evict_key]
        self.cache[key] = (value, 1)
        self.freq_map[1][key] = None
        self.min_freq = 1

class MyHashMap:
    def __init__(self, size=1000):
        self.size = size
        self.buckets = [[] for _ in range(size)]

        return key % self.size

        idx = self._hash(key)
        for i, (k, v) in enumerate(self.buckets[idx]):
            if k == key:
                self.buckets[idx][i] = (key, value)
                return
        self.buckets[idx].append((key, value))

        idx = self._hash(key)
        for k, v in self.buckets[idx]:
            if k == key:
                return v
        return -1

        idx = self._hash(key)
        self.buckets[idx] = [(k, v) for k, v in self.buckets[idx] if k != key]

        self.rows = len(data)
        self.cols = len(data[0]) if data else 0

        return '\n'.join([' '.join(f'{x:8.2f}' for x in row) for row in self.data])
    
    def __add__(self, other):
        return Matrix([[self.data[i][j] + other.data[i][j] 
                       for j in range(self.cols)] for i in range(self.rows)])
    
    def __mul__(self, other):
        if isinstance(other, (int, float)):
            return Matrix([[x * other for x in row] for row in self.data])
        result = [[sum(self.data[i][k] * other.data[k][j] for k in range(self.cols))
                   for j in range(other.cols)] for i in range(self.rows)]
        return Matrix(result)
    
    def transpose(self):
        return Matrix([[self.data[j][i] for j in range(self.rows)] for i in range(self.cols)])
    
    def determinant(self):
        if self.rows != self.cols:
            raise ValueError("Square matrix required")
        if self.rows == 1:
            return self.data[0][0]
        if self.rows == 2:
            return self.data[0][0]*self.data[1][1] - self.data[0][1]*self.data[1][0]
        det = 0
        for j in range(self.cols):
            minor = Matrix([[self.data[i][k] for k in range(self.cols) if k != j]
                           for i in range(1, self.rows)])
            det += ((-1)**j) * self.data[0][j] * minor.determinant()
        return det

import cmath

def fft(x):
    n = len(x)
    if n <= 1:
        return x
    even = fft(x[0::2])
    odd = fft(x[1::2])
    T = [cmath.exp(-2j * cmath.pi * k / n) * odd[k] for k in range(n // 2)]
    return [even[k] + T[k] for k in range(n // 2)] + \
           [even[k] - T[k] for k in range(n // 2)]

def ifft(x):
    n = len(x)
    conjugated = [v.conjugate() for v in x]
    result = fft(conjugated)
    return [v.conjugate() / n for v in result]

def linear_regression(x, y):
    n = len(x)
    sum_x = sum(x)
    sum_y = sum(y)
    sum_xy = sum(xi * yi for xi, yi in zip(x, y))
    sum_x2 = sum(xi**2 for xi in x)
    
    slope = (n * sum_xy - sum_x * sum_y) / (n * sum_x2 - sum_x**2)
    intercept = (sum_y - slope * sum_x) / n
    
    # R-squared
    y_pred = [slope * xi + intercept for xi in x]
    ss_res = sum((yi - yp)**2 for yi, yp in zip(y, y_pred))
    y_mean = sum_y / n
    ss_tot = sum((yi - y_mean)**2 for yi in y)
    r_squared = 1 - ss_res / ss_tot

        self.coeffs = list(coefficients)  # [a0, a1, a2, ...] = a0 + a1*x + a2*x^2 + ...

        for i, c in enumerate(self.coeffs):
            if c == 0: continue
            if i == 0: terms.append(f"{c}")
            elif i == 1: terms.append(f"{c}x")
            else: terms.append(f"{c}x^{i}")

        size = max(len(self.coeffs), len(other.coeffs))
        result = [0] * size
        for i, c in enumerate(self.coeffs): result[i] += c
        for i, c in enumerate(other.coeffs): result[i] += c

            for j, b in enumerate(other.coeffs):

        return Polynomial(result)
    
    def derivative(self):
        if len(self.coeffs) <= 1:
            return Polynomial([0])

    padding = data[-1]

    """Simplified substitution-permutation cipher (NOT secure — for learning)."""
    result = xor_bytes(block, key)
    # Simple substitution
    result = bytes((b + 47) % 256 for b in result)
    # Permutation (reverse)
    result = result[::-1]
    return result

    result = block[::-1]
    result = bytes((b - 47) % 256 for b in result)
    result = xor_bytes(result, key)
    return result

    padded = pad_pkcs7(plaintext.encode())
    blocks = [padded[i:i+16] for i in range(0, len(padded), 16)]
    ciphertext = iv
    prev = iv
    for block in blocks:
        xored = xor_bytes(block, prev)
        encrypted = simple_block_encrypt(xored, key)
        ciphertext += encrypted
        prev = encrypted

        prev = block

class MersenneTwister:
    """Simplified Mersenne Twister PRNG (MT19937)."""
    def __init__(self, seed=5489):
        self.n = 624
        self.m = 397
        self.mt = [0] * self.n
        self.index = self.n + 1
        self.seed(seed)
    
    def seed(self, s):
        self.mt[0] = s & 0xFFFFFFFF
        for i in range(1, self.n):
            self.mt[i] = (1812433253 * (self.mt[i-1] ^ (self.mt[i-1] >> 30)) + i) & 0xFFFFFFFF
        self.index = self.n
    
    def _generate(self):
        for i in range(self.n):
            y = (self.mt[i] & 0x80000000) + (self.mt[(i+1) % self.n] & 0x7FFFFFFF)
            self.mt[i] = self.mt[(i + self.m) % self.n] ^ (y >> 1)
            if y % 2:
                self.mt[i] ^= 0x9908B0DF
        self.index = 0
    
    def random_int(self):
        if self.index >= self.n:
            self._generate()
        y = self.mt[self.index]
        y ^= y >> 11
        y ^= (y << 7) & 0x9D2C5680
        y ^= (y << 15) & 0xEFC60000
        y ^= y >> 18
        self.index += 1
        return y & 0xFFFFFFFF
    
    def random(self):
        return self.random_int() / 0xFFFFFFFF

                       for j in range(self.cols)] for i in range(self.rows)])

        result = [[sum(self.data[i][k] * other.data[k][j]
                       for k in range(self.cols))
                   for j in range(other.cols)] for i in range(self.rows)]
        return Matrix(result)

                       for i in range(self.cols)])

            raise ValueError("Must be square")

def is_prime_miller_rabin(n, k=10):
    if n < 2: return False
    if n == 2 or n == 3: return True

    d, r = n - 1, 0
    while d % 2 == 0:

        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1: break
        else:
            return False
    return True

        return json.loads(base64.urlsafe_b64decode(data + "=" * padding))

        header = {"alg": "HS256", "typ": "JWT"}
        payload["iat"] = int(time.time())

        payload_b64 = self._b64encode(payload)
        
        signature = hmac.new(
            self.secret.encode(),
            f"{header_b64}.{payload_b64}".encode(),

        if len(parts) != 3:

            self.secret.encode(),
            f"{header_b64}.{payload_b64}".encode(),
            hashlib.sha256
        ).hexdigest()
        
        if not hmac.compare_digest(signature, expected_sig):
            raise ValueError("Invalid signature")
        
        return self._b64decode(payload_b64)

def to_base(number, base):
    if number == 0:
        return "0"
    digits = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"

class Fraction:
    def __init__(self, num, den=1):
        if den == 0:
            raise ValueError("Zero denominator")
        g = gcd(abs(num), abs(den))
        sign = -1 if (num < 0) ^ (den < 0) else 1
        self.num = sign * abs(num) // g
        self.den = abs(den) // g

        return Fraction(self.num * other.den + other.num * self.den, self.den * other.den)
    
    def __sub__(self, other):
        return Fraction(self.num * other.den - other.num * self.den, self.den * other.den)
    
    def __mul__(self, other):
        return Fraction(self.num * other.num, self.den * other.den)
    
    def __truediv__(self, other):
        return Fraction(self.num * other.den, self.den * other.num)
    
    def __repr__(self):
        return f"{self.num}/{self.den}" if self.den != 1 else str(self.num)

class LinearRegression:
    def __init__(self):
        self.slope = 0
        self.intercept = 0
    
    def fit(self, X, y):
        n = len(X)
        x_mean = sum(X) / n
        y_mean = sum(y) / n
        
        numerator = sum((X[i] - x_mean) * (y[i] - y_mean) for i in range(n))
        denominator = sum((X[i] - x_mean)**2 for i in range(n))
        
        self.slope = numerator / denominator
        self.intercept = y_mean - self.slope * x_mean
        
        # R-squared
        y_pred = [self.predict(x) for x in X]
        ss_res = sum((y[i] - y_pred[i])**2 for i in range(n))
        ss_tot = sum((y[i] - y_mean)**2 for i in range(n))
        self.r_squared = 1 - ss_res / ss_tot
    
    def predict(self, x):
        return self.slope * x + self.intercept

def distance(p1, p2):
    return math.sqrt(sum((a-b)**2 for a, b in zip(p1, p2)))

def kmeans(points, k, max_iter=100):
    centroids = random.sample(points, k)
    
    for _ in range(max_iter):
        clusters = [[] for _ in range(k)]
        for point in points:
            distances = [distance(point, c) for c in centroids]
            cluster_idx = distances.index(min(distances))
            clusters[cluster_idx].append(point)
        
        new_centroids = []
        for cluster in clusters:
            if cluster:
                centroid = tuple(sum(dim)/len(cluster) for dim in zip(*cluster))
                new_centroids.append(centroid)
            else:
                new_centroids.append(random.choice(points))
        
        if new_centroids == centroids:
            break
        centroids = new_centroids
    
    return centroids, clusters

import time

class TokenBucket:
    def __init__(self, rate, capacity):
        self.rate = rate          # Tokens per second
        self.capacity = capacity  # Max tokens

def bfs_path(graph, start, goal):
    queue = deque([(start, [start])])
    seen = {start}
    while queue:
        node, path = queue.popleft()
        if node == goal:
            return path
        for nxt in graph[node]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append((nxt, path + [nxt]))
    return None

Program 143: Dijkstra with a Heap

import heapq

graph = {
    "A": {"B": 1, "C": 4},
    "B": {"C": 2, "D": 5},
    "C": {"D": 1},
    "D": {},
}

def dijkstra(graph, start):
    dist = {n: float("inf") for n in graph}
    dist[start] = 0
    heap = [(0, start)]
    while heap:
        d, node = heapq.heappop(heap)
        if d > dist[node]:
            continue                    # a stale entry: we already did better
        for nxt, weight in graph[node].items():
            nd = d + weight
            if nd < dist[nxt]:
                dist[nxt] = nd
                heapq.heappush(heap, (nd, nxt))
    return dist

print(dijkstra(graph, "A"))
# Output: {'A': 0, 'B': 1, 'C': 3, 'D': 4}

# A-C directly is 4, but A-B-C is 1+2 = 3, so D comes out at 4 rather than 5.

Program 146: The Knapsack, Bottom Up

def knapsack(weights, values, capacity):
    n = len(weights)
    table = [[0] * (capacity + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for c in range(capacity + 1):
            table[i][c] = table[i - 1][c]                     # skip item i
            if weights[i - 1] <= c:                           # or take it
                take = table[i - 1][c - weights[i - 1]] + values[i - 1]
                table[i][c] = max(table[i][c], take)
    return table[n][capacity]

print(knapsack([2], [10], 1))                      # Output: 0

# 9 = items of weight 3 and 4 (values 4 + 5), which beats the weight-5 item.

Program 147: Longest Common Subsequence

def lcs(a, b):
    m, n = len(a), len(b)
    table = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                table[i][j] = table[i - 1][j - 1] + 1
            else:
                table[i][j] = max(table[i - 1][j], table[i][j - 1])
    # walk the table backwards to recover the subsequence itself
    out, i, j = [], m, n
    while i and j:
        if a[i - 1] == b[j - 1]:
            out.append(a[i - 1]); i -= 1; j -= 1
        elif table[i - 1][j] >= table[i][j - 1]:
            i -= 1
        else:
            j -= 1
    return table[m][n], "".join(reversed(out))

print(lcs("ABCBDAB", "BDCABA"))       # Output: (4, 'BCBA')
print(lcs("abc", "xyz"))              # Output: (0, '')

def edit_distance(a, b):
    prev = list(range(len(b) + 1))
    for i, ca in enumerate(a, 1):
        cur = [i]
        for j, cb in enumerate(b, 1):
            cur.append(min(
                prev[j] + 1,                       # delete
                cur[j - 1] + 1,                    # insert
                prev[j - 1] + (ca != cb),          # substitute
            ))
        prev = cur
    return prev[-1]

Program 150: Union-Find with Path Compression

class DisjointSet:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n

    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]    # path compression
            x = self.parent[x]
        return x

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False                # already connected: this edge is a cycle
        if self.rank[ra] < self.rank[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        if self.rank[ra] == self.rank[rb]:
            self.rank[ra] += 1
        return True

ds = DisjointSet(6)
for a, b in [(0, 1), (1, 2), (3, 4)]:
    ds.union(a, b)
print(ds.find(0) == ds.find(2))     # Output: True
print(ds.find(0) == ds.find(3))     # Output: False
print(ds.union(0, 2))               # Output: False

        for ch in word:
            node = node.setdefault(ch, {})

                if k == "$":
                    out.append(prefix + acc)
                else:

        walk(node, "")
        return out

Program 157: Aho-Corasick: find every keyword in one pass

"""Aho-Corasick builds a trie of all the patterns, then adds a "failure link" from
each node to the longest proper suffix that is also a prefix of some pattern - the
same idea as the KMP failure function, generalised to many patterns at once.

One scan of the text finds every occurrence of every pattern:
    naive multi-pattern : O(len(text) * sum(len(p)))
    Aho-Corasick        : O(sum(len(p)) + len(text) + matches)

This is the algorithm behind keyword filters, PII redaction and blocklists, where
you need hundreds of terms checked against every document.
"""
from collections import deque


class AhoCorasick:
    def __init__(self, patterns):
        # goto[state][char] -> state ; out[state] -> patterns ending here
        self.goto = [{}]
        self.out = [[]]
        self.fail = [0]
        for p in patterns:
            self._add(p)
        self._build_failure_links()

    def _add(self, pattern):
        state = 0
        for ch in pattern:
            if ch not in self.goto[state]:
                self.goto.append({})
                self.out.append([])
                self.fail.append(0)
                self.goto[state][ch] = len(self.goto) - 1
            state = self.goto[state][ch]
        self.out[state].append(pattern)

    def _build_failure_links(self):
        q = deque()
        for ch, nxt in self.goto[0].items():
            self.fail[nxt] = 0              # depth-1 nodes fail back to the root
            q.append(nxt)
        while q:
            state = q.popleft()
            for ch, nxt in self.goto[state].items():
                q.append(nxt)
                f = self.fail[state]
                while f and ch not in self.goto[f]:
                    f = self.fail[f]
                self.fail[nxt] = self.goto[f].get(ch, 0)
                # inherit the matches of the node we fail to - "he" inside "she"
                self.out[nxt] += self.out[self.fail[nxt]]

    def search(self, text):
        state, hits = 0, []
        for i, ch in enumerate(text):
            while state and ch not in self.goto[state]:
                state = self.fail[state]
            state = self.goto[state].get(ch, 0)
            for pattern in self.out[state]:
                hits.append((i - len(pattern) + 1, pattern))
        return hits


patterns = ["he", "she", "his", "hers"]
text = "ushers hit his shed"

ac = AhoCorasick(patterns)
found = ac.search(text)
print("text     :", text)
print("patterns :", patterns)
for pos, pat in found:
    print("   at %2d  %-5s %s" % (pos, pat, " " * pos + pat))

# The overlapping matches are the point: "she", "he" and "hers" all sit inside "ushers".
print("total matches:", len(found))

# Same answers as the naive scan, but the naive one restarts at every position
naive = sorted((i, p) for p in patterns
               for i in range(len(text)) if text.startswith(p, i))
print("agrees with naive scan:", sorted(found) == naive)
print("trie states built:", len(ac.goto))

Q45 · Find the maximum sum subarray (Kadane's Algorithm)

"""
Find the contiguous subarray with the largest sum.
Kadane's Algorithm: Track current max and global max.
Time Complexity: O(n)
Space Complexity: O(1)
"""


def max_subarray_sum(lst):
    """Kadane's Algorithm for maximum subarray sum."""
    if not lst:
        return 0

    max_current = max_global = lst[0]
    start = end = temp_start = 0

    for i in range(1, len(lst)):
        if lst[i] > max_current + lst[i]:
            max_current = lst[i]
            temp_start = i
        else:
            max_current += lst[i]

        if max_current > max_global:
            max_global = max_current
            start = temp_start
            end = i

    return max_global, lst[start:end + 1]


# Examples
print("Q45: Maximum Sum Subarray (Kadane's)")
print(max_subarray_sum([-2, 1, -3, 4, -1, 2, 1, -5, 4]))  # (6, [4, -1, 2, 1])
print(max_subarray_sum([1, 2, 3, -2, 5]))  # (9, [1, 2, 3, -2, 5])
print()

Boyer-Moore Voting Algorithm.
Time Complexity: O(n)
Space Complexity: O(1)
"""


def majority_element(lst):
    """Boyer-Moore Voting Algorithm."""
    candidate = None
    count = 0

    for num in lst:
        if count == 0:
            candidate = num
            count = 1
        elif num == candidate:
            count += 1
        else:
            count -= 1

    # Verify the candidate is actually majority
    if lst.count(candidate) > len(lst) // 2:
        return candidate
    return None

Q49 · Find the kth largest element

"""
Multiple approaches:
1. Sort and index - O(n log n)
2. Min-heap of size k - O(n log k)
3. Quickselect - O(n) average
"""
import heapq


def kth_largest(lst, k):
    """Using a min-heap of size k."""
    return heapq.nlargest(k, lst)[-1]


def kth_largest_sort(lst, k):
    """Using sorting."""
    return sorted(lst, reverse=True)[k - 1]


def kth_largest_quickselect(lst, k):
    """Quickselect algorithm - O(n) average."""
    if not lst or k > len(lst):
        return None
    # Convert to finding (n-k)th smallest
    target = len(lst) - k

    def quickselect(arr, left, right, target_idx):
        pivot = arr[right]
        store = left
        for i in range(left, right):
            if arr[i] < pivot:
                arr[i], arr[store] = arr[store], arr[i]
                store += 1
        arr[store], arr[right] = arr[right], arr[store]

        if store == target_idx:
            return arr[store]
        elif store < target_idx:
            return quickselect(arr, store + 1, right, target_idx)
        else:
            return quickselect(arr, left, store - 1, target_idx)

    return quickselect(lst[:], 0, len(lst) - 1, target)


# Examples

print(kth_largest([3, 2, 1, 5, 6, 4], 2))  # 5
print(kth_largest_quickselect([3, 2, 3, 1, 2, 4, 5, 5, 6], 4))  # 4

Q52 · Implement bubble sort

"""
Repeatedly swap adjacent elements if they are in the wrong order.
Optimization: Stop early if no swaps occur in a pass.
Time Complexity: O(n^2) worst, O(n) best (already sorted)
Space Complexity: O(1)
"""


def bubble_sort(lst):
    """Optimized bubble sort with early termination."""
    arr = lst[:]
    n = len(arr)
    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                swapped = True
        if not swapped:
            break  # Array is already sorted
    return arr

"""
Build sorted array one element at a time by inserting into correct position.
Time Complexity: O(n^2) worst, O(n) best (nearly sorted)
Space Complexity: O(1)
"""


def insertion_sort(lst):
    """Insertion sort - insert each element into sorted portion."""
    arr = lst[:]
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
    return arr

Q55 · Implement merge sort

"""
Divide and conquer: split array in half, sort each half, merge.
Time Complexity: O(n log n) always
Space Complexity: O(n)
"""


def merge_sort(lst):
    """Merge sort - divide, sort, merge."""
    if len(lst) <= 1:
        return lst

    mid = len(lst) // 2
    left = merge_sort(lst[:mid])
    right = merge_sort(lst[mid:])

    return merge(left, right)


def merge(left, right):
    """Merge two sorted lists."""
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result


# Examples

"""
Divide and conquer: choose pivot, partition into less/greater, sort recursively.
Time Complexity: O(n log n) average, O(n^2) worst
Space Complexity: O(log n) for recursion stack
"""


def quick_sort(lst):
    """Quick sort with last element as pivot."""
    if len(lst) <= 1:
        return lst

    pivot = lst[-1]
    left = [x for x in lst[:-1] if x <= pivot]
    right = [x for x in lst[:-1] if x > pivot]

    return quick_sort(left) + [pivot] + quick_sort(right)


def quick_sort_inplace(lst, low=0, high=None):
    """In-place quick sort using Lomuto partition."""
    if high is None:
        high = len(lst) - 1
    if low < high:
        pivot_idx = partition(lst, low, high)
        quick_sort_inplace(lst, low, pivot_idx - 1)
        quick_sort_inplace(lst, pivot_idx + 1, high)
    return lst


def partition(lst, low, high):
    """Lomuto partition scheme."""
    pivot = lst[high]
    i = low - 1
    for j in range(low, high):
        if lst[j] <= pivot:
            i += 1
            lst[i], lst[j] = lst[j], lst[i]
    lst[i + 1], lst[high] = lst[high], lst[i + 1]
    return i + 1


# Examples
print("Q56: Quick Sort")
print(quick_sort([10, 7, 8, 9, 1, 5]))  # [1, 5, 7, 8, 9, 10]
arr = [10, 7, 8, 9, 1, 5]
print(quick_sort_inplace(arr))  # [1, 5, 7, 8, 9, 10]

Q651 · Longest Substring Without Repeating Characters

"""
Find the longest substring that doesn't have repeating characters.
Example: "abcdefabcde" → "abcdef" (length 6)

Approach: Sliding window with set
- Use a set to track characters in current window
- If duplicate found, shrink window from left until duplicate is gone
- Track the longest valid window

Time Complexity: O(n)
Space Complexity: O(min(n, charset_size))
"""


def longest_substring_without_repeating(s):
    """Find longest substring without repeating characters."""
    best_str = ""
    temp = ""

    for i in range(len(s)):
        temp += s[i]

        # if duplicate exists in current window
        while len(temp) != len(set(temp)):
            temp = temp[1:]   # shrink from left

        if len(temp) > len(best_str):
            best_str = temp

    return best_str

Q652 · String Compression (Run-Length Encoding)

"""
Compress a string by counting consecutive identical characters.
Example: "aaabbbcc" → "a3b3c2"

Approach: Single pass through the string
- Count consecutive characters
- Build result string with character + count
- Optimization: Only add count if > 1 (optional)

Time Complexity: O(n)
Space Complexity: O(n)
"""


def compress_string(s):
    """Compress string using run-length encoding."""
    count = 1
    final_result = ""

    for i in range(len(s) - 1):
        if s[i] == s[i + 1]:
            count += 1
        else:
            final_result += s[i] + str(count)
            count = 1

    final_result += s[-1] + str(count)

    return final_result


def compress_string_optimized(s):
    """Compress string - only add count if > 1."""
    if not s:
        return ""

    count = 1
    final_result = ""

    for i in range(len(s) - 1):
        if s[i] == s[i + 1]:
            count += 1
        else:
            final_result += s[i]
            if count > 1:
                final_result += str(count)
            count = 1

    final_result += s[-1]
    if count > 1:
        final_result += str(count)

    return final_result
