data = [(1, 3.14, b'Hi'), (2, 2.72, b'By')]

with open("data.bin", "wb") as f:
    for id_, val, msg in data:
        packed = struct.pack("if2s", id_, val, msg)
        f.write(packed)

# Read binary data
with open("data.bin", "rb") as f:
    size = struct.calcsize("if2s")
    while True:
        chunk = f.read(size)
        if not chunk:
            break
        id_, val, msg = struct.unpack("if2s", chunk)
        print(f"ID={id_}, Val={val:.2f}, Msg={msg.decode()}")

Program 10: File Watcher

import os
import time
import hashlib

class FileWatcher:
    def __init__(self, path):
        self.path = path
        self.file_hashes = {}
        self._scan()
    
    def _hash_file(self, filepath):
        with open(filepath, 'rb') as f:
            return hashlib.md5(f.read()).hexdigest()
    
    def _scan(self):
        for root, dirs, files in os.walk(self.path):
            for name in files:
                fp = os.path.join(root, name)
                self.file_hashes[fp] = self._hash_file(fp)
    
    def check_changes(self):
        changes = {"modified": [], "added": [], "deleted": []}
        current = {}
        for root, dirs, files in os.walk(self.path):
            for name in files:
                fp = os.path.join(root, name)
                h = self._hash_file(fp)
                current[fp] = h
                if fp not in self.file_hashes:
                    changes["added"].append(fp)
                elif self.file_hashes[fp] != h:
                    changes["modified"].append(fp)
        for fp in self.file_hashes:
            if fp not in current:
                changes["deleted"].append(fp)
        self.file_hashes = current
        return changes

print("FileWatcher class implemented")

def parse_env_file(filepath):
    env = {}
    try:
        with open(filepath, 'r') as f:
            for line in f:
                line = line.strip()
                if not line or line.startswith('#'):
                    continue
                if '=' in line:
                    key, value = line.split('=', 1)
                    value = value.strip().strip('"').strip("'")
                    env[key.strip()] = value
    except FileNotFoundError:
        pass
    return env

Program 20: Streaming Large File Processing

def process_large_file(filepath, chunk_size=8192):
    """Process a large file in chunks without loading entirely into memory."""
    total_bytes = 0
    line_count = 0
    word_count = 0
    
    with open(filepath, 'r', buffering=chunk_size) as f:
        for line in f:
            total_bytes += len(line.encode())
            line_count += 1
            word_count += len(line.split())
    
    return {
        "bytes": total_bytes,
        "lines": line_count,
        "words": word_count
    }

# Create sample file
with open("large_sample.txt", "w") as f:
    for i in range(1000):
        f.write(f"Line {i}: This is sample data for testing\n")

# Output: {'bytes': ~45000, 'lines': 1000, 'words': 8000}

Program 23: Protobuf-Style Serialization

import struct

class SimpleProto:
    """Minimal binary serialization format."""
    
    @staticmethod
    def encode(data: dict) -> bytes:
        parts = []
        for key, value in data.items():
            key_bytes = key.encode()
            parts.append(struct.pack("H", len(key_bytes)))
            parts.append(key_bytes)
            if isinstance(value, int):
                parts.append(b'\x01')
                parts.append(struct.pack("q", value))
            elif isinstance(value, str):
                val_bytes = value.encode()
                parts.append(b'\x02')
                parts.append(struct.pack("I", len(val_bytes)))
                parts.append(val_bytes)
            elif isinstance(value, float):
                parts.append(b'\x03')
                parts.append(struct.pack("d", value))
        return b''.join(parts)
    
    @staticmethod
    def decode(data: bytes) -> dict:
        result = {}
        offset = 0
        while offset < len(data):
            key_len = struct.unpack_from("H", data, offset)[0]; offset += 2
            key = data[offset:offset+key_len].decode(); offset += key_len
            type_byte = data[offset]; offset += 1
            if type_byte == 1:
                result[key] = struct.unpack_from("q", data, offset)[0]; offset += 8
            elif type_byte == 2:
                val_len = struct.unpack_from("I", data, offset)[0]; offset += 4
                result[key] = data[offset:offset+val_len].decode(); offset += val_len
            elif type_byte == 3:
                result[key] = struct.unpack_from("d", data, offset)[0]; offset += 8
        return result

original = {"name": "Alice", "age": 30, "score": 95.5}
encoded = SimpleProto.encode(original)
decoded = SimpleProto.decode(encoded)
print(f"Original: {original}")
print(f"Encoded: {len(encoded)} bytes")
print(f"Decoded: {decoded}")

Program 24: Parquet-Like Column Store

import json

class ColumnStore:
    """Simple columnar storage format."""
    
    def __init__(self):
        self.columns = {}
        self.row_count = 0
    
    def from_rows(self, rows):
        if not rows:
            return
        self.row_count = len(rows)
        for key in rows[0]:
            self.columns[key] = [row.get(key) for row in rows]
    
    def select(self, *cols):
        result = []
        for i in range(self.row_count):
            row = {col: self.columns[col][i] for col in cols if col in self.columns}
            result.append(row)
        return result
    
    def where(self, col, predicate):
        indices = [i for i, v in enumerate(self.columns[col]) if predicate(v)]
        result = {}
        for key in self.columns:
            result[key] = [self.columns[key][i] for i in indices]
        store = ColumnStore()
        store.columns = result
        store.row_count = len(indices)
        return store
    
    def aggregate(self, col, func):
        return func(self.columns[col])

rows = [
    {"name": "Alice", "dept": "Eng", "salary": 95000},
    {"name": "Bob", "dept": "Mkt", "salary": 75000},
    {"name": "Charlie", "dept": "Eng", "salary": 88000},
]

cs = ColumnStore()
cs.from_rows(rows)

eng = cs.where("dept", lambda x: x == "Eng")

Program 25: File Encryption/Decryption

import hashlib
import os

def xor_encrypt(data: bytes, key: bytes) -> bytes:
    """Simple XOR encryption (for learning — use AES in production)."""
    key_repeated = (key * (len(data) // len(key) + 1))[:len(data)]
    return bytes(a ^ b for a, b in zip(data, key_repeated))

def encrypt_file(input_path, output_path, password):
    key = hashlib.sha256(password.encode()).digest()
    salt = os.urandom(16)
    derived_key = hashlib.pbkdf2_hmac('sha256', key, salt, 100000)
    
    with open(input_path, 'rb') as f:
        plaintext = f.read()
    
    ciphertext = xor_encrypt(plaintext, derived_key)
    
    with open(output_path, 'wb') as f:
        f.write(salt + ciphertext)

def decrypt_file(input_path, output_path, password):
    key = hashlib.sha256(password.encode()).digest()
    
    with open(input_path, 'rb') as f:
        salt = f.read(16)
        ciphertext = f.read()
    
    derived_key = hashlib.pbkdf2_hmac('sha256', key, salt, 100000)
    plaintext = xor_encrypt(ciphertext, derived_key)
    
    with open(output_path, 'wb') as f:
        f.write(plaintext)

    f.write("Top secret message!")
