Q445 · Transactions

class TransactionDemo:
    """Simulates database transactions."""
    def __init__(self):
        self.data = {}
        self._backup = None

    def begin(self):
        self._backup = self.data.copy()

    def commit(self):
        self._backup = None

    def rollback(self):
        if self._backup is not None:
            self.data = self._backup
            self._backup = None

print("Q445: Transactions")
db = TransactionDemo()
db.begin()
db.data["user"] = "Alice"
db.data["order"] = 123
print(f"  Before rollback: {db.data}")
db.rollback()
print(f"  After rollback: {db.data}")

db.begin()
db.data["user"] = "Bob"
db.commit()
print(f"  After commit: {db.data}")
print()

Q453 · Composite indexes

"""
Composite indexes cover multiple columns. Column order matters!

Rule: Put most selective column first, or follow query pattern.
A composite index on (a, b, c) can satisfy:
- WHERE a = 1
- WHERE a = 1 AND b = 2
- WHERE a = 1 AND b = 2 AND c = 3
But NOT: WHERE b = 2 (needs leftmost prefix)

This is called the "leftmost prefix" rule.
"""

print("Q453: Composite indexes")

print("  Satisfies: WHERE user_id=1 AND status='paid'")
print("  Satisfies: WHERE user_id=1 AND status='paid' AND created_at > '2024-01-01'")
print("  NOT:       WHERE status='paid'  (skips leftmost column)")

Q495 · Circuit breaker pattern

print("Q495: Circuit breaker pattern")
print("  Prevent cascading failures by stopping calls to failing services.")
print("  States: CLOSED (normal) → OPEN (failing) → HALF-OPEN (testing)")

class CircuitBreaker:
    def __init__(self, failure_threshold=3, reset_timeout=10):
        self.failure_count = 0
        self.threshold = failure_threshold
        self.state = "CLOSED"

    def call(self, func, *args):
        if self.state == "OPEN":
            raise RuntimeError("Circuit is OPEN")
        try:
            result = func(*args)
            self.failure_count = 0
            return result
        except Exception:
            self.failure_count += 1
            if self.failure_count >= self.threshold:
                self.state = "OPEN"
            raise

cb = CircuitBreaker(failure_threshold=2)
for i in range(3):
    try:
        cb.call(lambda: 1/0)
    except (ZeroDivisionError, RuntimeError) as e:
        print(f"  Attempt {i+1}: {type(e).__name__} (state={cb.state})")
print()

Output 

Q495: Circuit breaker pattern
  Prevent cascading failures by stopping calls to failing services.
  States: CLOSED (normal) → OPEN (failing) → HALF-OPEN (testing)
  Attempt 1: ZeroDivisionError (state=CLOSED)
  Attempt 2: ZeroDivisionError (state=OPEN)
  Attempt 3: RuntimeError (state=OPEN)

class SimpleTaskQueue:
    def __init__(self):
        self.queues = {"high": [], "default": [], "low": []}

    def enqueue(self, task, priority="default"):
        self.queues[priority].append(task)

    def dequeue(self):
        for priority in ["high", "default", "low"]:
            if self.queues[priority]:
                return self.queues[priority].pop(0)
        return None

tq = SimpleTaskQueue()
tq.enqueue("send_email", "high")
tq.enqueue("generate_report", "low")
tq.enqueue("process_payment", "high")
print(f"  Next task: {tq.dequeue()}")  # high priority first
