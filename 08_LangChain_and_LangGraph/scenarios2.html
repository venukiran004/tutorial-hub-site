from langchain_core.messages import HumanMessage, AIMessage

class SummaryMemory:
    """Summarize old messages to keep context window manageable."""
    
    def __init__(self, llm, max_messages: int = 10):
        self.llm = llm
        self.max_messages = max_messages
        self.messages = []
        self.summary = ""
    
    def add_messages(self, human_msg: str, ai_msg: str):
        self.messages.append(HumanMessage(content=human_msg))
        self.messages.append(AIMessage(content=ai_msg))
        
        if len(self.messages) > self.max_messages:
            self._summarize_old_messages()
    
    def _summarize_old_messages(self):
        old_msgs = self.messages[:self.max_messages // 2]
        old_text = "\n".join(f"{'Human' if isinstance(m, HumanMessage) else 'AI'}: {m.content}" for m in old_msgs)
        
        summary_prompt = f"Summarize this conversation:\n{old_text}\n\nPrevious summary: {self.summary}"
        self.summary = self.llm.invoke(summary_prompt).content
        self.messages = self.messages[self.max_messages // 2:]
    
    def get_context(self) -> list:
        context = []
        if self.summary:
            context.append(("system", f"Conversation summary: {self.summary}"))
        context.extend([(type(m).__name__.lower().replace("message", ""), m.content) for m in self.messages])
        return context

Program 38: Window Memory with Token Limit

from langchain_core.messages import BaseMessage
import tiktoken

class TokenWindowMemory:
    """Keep only recent messages that fit within token budget."""
    
    def __init__(self, max_tokens: int = 4000, model: str = "gpt-4o-mini"):
        self.max_tokens = max_tokens
        self.encoder = tiktoken.encoding_for_model(model)
        self.messages: list[BaseMessage] = []
    
    def count_tokens(self, messages: list[BaseMessage]) -> int:
        return sum(len(self.encoder.encode(m.content)) for m in messages)
    
    def add_message(self, message: BaseMessage):
        self.messages.append(message)
        # Trim oldest messages until within budget
        while self.count_tokens(self.messages) > self.max_tokens and len(self.messages) > 2:
            self.messages.pop(0)
    
    def get_messages(self) -> list[BaseMessage]:
        return self.messages.copy()

from langgraph.graph import StateGraph, END
from typing import Literal

class MultiAgentState(TypedDict):
    messages: Annotated[list, add]
    next_agent: str

def supervisor(state: MultiAgentState) -> dict:
    """Decide which agent should handle the task."""
    last_message = state["messages"][-1]
    
    decision = llm.with_structured_output(
        type("Route", (), {"__annotations__": {"next": Literal["researcher", "coder", "writer", "FINISH"]}})
    ).invoke(
        f"Based on the conversation, who should act next? "
        f"Choose: researcher (for facts), coder (for code), writer (for text), FINISH (if done).\n"
        f"Last message: {last_message}"
    )
    return {"next_agent": decision.next}

def researcher(state: MultiAgentState) -> dict:
    response = llm.invoke(
        "You are a research specialist. " + state["messages"][-1]
    ).content
    return {"messages": [f"[Researcher]: {response}"]}

def coder(state: MultiAgentState) -> dict:
    response = llm.invoke(
        "You are a Python expert. Write clean, documented code. " + state["messages"][-1]
    ).content
    return {"messages": [f"[Coder]: {response}"]}

def writer(state: MultiAgentState) -> dict:
    response = llm.invoke(
        "You are a technical writer. Write clear, concise content. " + state["messages"][-1]
    ).content
    return {"messages": [f"[Writer]: {response}"]}

def route_agent(state: MultiAgentState) -> str:
    return state["next_agent"]

graph = StateGraph(MultiAgentState)
graph.add_node("supervisor", supervisor)
graph.add_node("researcher", researcher)
graph.add_node("coder", coder)
graph.add_node("writer", writer)

    "researcher": "researcher",
    "coder": "coder",
    "writer": "writer",
    "FINISH": END,

Program 48: Error Recovery Agent

from langgraph.graph import StateGraph, END

class ResilientState(TypedDict):
    task: str
    result: str
    error: str
    retry_count: int
    status: str

def execute_task(state: ResilientState) -> dict:
    """Try to execute the task."""
    try:
        result = llm.invoke(state["task"]).content
        return {"result": result, "status": "success", "error": ""}
    except Exception as e:
        return {
            "error": str(e),
            "status": "error",
            "retry_count": state["retry_count"] + 1,
        }

def handle_error(state: ResilientState) -> dict:
    """Analyze error and adjust strategy."""
    error_analysis = llm.invoke(
        f"The task '{state['task']}' failed with error: {state['error']}. "
        f"Suggest a modified approach."
    ).content
    return {"task": error_analysis, "status": "retrying"}

def should_retry(state: ResilientState) -> str:
    if state["status"] == "success":
        return "done"
    if state["retry_count"] >= 3:
        return "give_up"
    return "retry"

def give_up(state: ResilientState) -> dict:
    return {"result": f"Failed after {state['retry_count']} attempts. Last error: {state['error']}"}

graph = StateGraph(ResilientState)
graph.add_node("execute", execute_task)
graph.add_node("handle_error", handle_error)
graph.add_node("give_up", give_up)

graph.set_entry_point("execute")
graph.add_conditional_edges("execute", should_retry, {
    "done": END,
    "retry": "handle_error",
    "give_up": "give_up",
})
graph.add_edge("handle_error", "execute")  # Retry after error handling
graph.add_edge("give_up", END)

app = graph.compile()
